📝 Mã nguồn BFS
function bfs(graph, start) {
const visited = new Set([start]);
const queue = [start];
const dist = {[start]: 0};
while (queue.length > 0) {
// Dequeue từ FRONT (FIFO)
const node = queue.shift();
process(node); // Level-order
for (const nb of graph[node]) {
if (!visited.has(nb)) {
visited.add(nb);
dist[nb] = dist[node] + 1;
queue.push(nb); // Enqueue
}
}
}
return dist;
}
